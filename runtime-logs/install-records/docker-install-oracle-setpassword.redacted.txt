-- 创建永久表空间，初始大小500MB，自动扩展每次50MB，最大2GB
CREATE TABLESPACE YSSP_data DATAFILE '/opt/oracle/oradata/YSSP_data01.dbf'
    SIZE 500M AUTOEXTEND ON NEXT 50M MAXSIZE 2G EXTENT MANAGEMENT LOCAL SEGMENT SPACE MANAGEMENT AUTO;
-- 创建临时表空间（用于排序操作）
CREATE TEMPORARY TABLESPACE YSSP_temp_data TEMPFILE '/opt/oracle/oradata/YSSP_temp_data01.dbf'
    SIZE 100M AUTOEXTEND ON NEXT 20M MAXSIZE 500M;


-- 创建用户并分配表空间
CREATE USER YSSP IDENTIFIED BY "<由运行环境注入>"
    DEFAULT TABLESPACE YSSP_data TEMPORARY TABLESPACE YSSP_temp_data QUOTA 1G ON YSSP_data; -- 分配1GB配额
-- 允许用户无限制使用表空间
ALTER USER YSSP QUOTA UNLIMITED ON YSSP_data;

重置序列值
-- 假设当前序列值为100，目标为365
ALTER SEQUENCE seq_sys_user INCREMENT BY 360;
SELECT seq_sys_user.NEXTVAL FROM DUAL;  -- 返回365
ALTER SEQUENCE seq_sys_user INCREMENT BY 1;
